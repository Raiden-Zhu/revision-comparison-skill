% Visual theme for a standalone comparison wrapper. Load annotations.tex first.
% Page size, margins, source packages, and equation identities belong to the wrapper.
\RequirePackage[T1]{fontenc}
\RequirePackage{lmodern}
\RequirePackage{tcolorbox}
\tcbuselibrary{breakable,skins}
\RequirePackage{fancyhdr}
\definecolor{rcInk}{HTML}{172033}
\definecolor{rcMuted}{HTML}{48566B}
\definecolor{rcPaper}{HTML}{FAFBFD}
\definecolor{rcBorder}{HTML}{DBE1EB}
\definecolor{rcContextBackground}{HTML}{EAF1EB}
\renewcommand{\familydefault}{\sfdefault}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.45em}
\setlength{\headheight}{14pt}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyfoot[L]{\sffamily\footnotesize\color{rcMuted}Revision Comparison}
\fancyfoot[R]{\sffamily\footnotesize\color{rcMuted}\thepage}
\newcommand{\rcReportTitle}[2]{%
  {\rmfamily\bfseries\fontsize{29}{34}\selectfont\color{rcInk}#1\par}
  \vspace{3pt}{\sffamily\large\color{rcMuted}#2\par}
  \vspace{7pt}{\color{rcRewritten}\rule{42pt}{2pt}}\par\vspace{4pt}}
\tcbset{rcCard/.style={enhanced,breakable,boxrule=0.5pt,arc=2mm,
  colframe=rcBorder,colback=white,colbacktitle=rcPaper,coltitle=rcInk,
  fonttitle=\sffamily\bfseries\small,left=10pt,right=10pt,top=6pt,bottom=6pt,
  toptitle=6pt,bottomtitle=6pt,before skip=8pt,after skip=8pt,
  pad at break*=2mm}}
\newtcolorbox{rcContext}[1]{rcCard,colback=rcContextBackground,
  colbacktitle=rcContextBackground,coltitle=rcNote,
  title={#1},title after break={#1\space (continued)}}
\newtcolorbox{rcPassage}[3]{rcCard,title={#1\space / #2\hfill #3},
  title after break={#1\space / #2\space (continued)\hfill #3}}
